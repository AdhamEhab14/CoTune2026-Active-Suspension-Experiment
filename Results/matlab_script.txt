

Ms = 2.45;     % Sprung mass [kg]
Mus = 1.00;    % Unsprung mass [kg]
Ks = 900;      % Suspension stiffness [N/m]
Kus = 1250;    % Tire stiffness [N/m]
Bs = 58.08;    % Suspension damping [N*s/m]
Bus = 14.726;  % Tire damping [N*s/m]



Kt  = 0.05667;        % Motor torque constant [N.m/A]
R_m = 3;   % Motor armature resistance [Ohms]
Kg  = 2.0;        % Gearbox ratio (2:1 pulley reduction) [dimensionless]
r   = 0.006;      % Capstan pulley radius [m] (6mm)
V_batt = 12;      % Battery Voltage supplied to the Cytron Driver [V]

% 2. RELATIVE STATE-SPACE REPRESENTATION
% States: x = [zs - zus; zus - zr; zs_dot; zus_dot]
A_rel = [0, 0, 1, -1;
    0, 0, 0, 1;
    -Ks/Ms, 0, -Bs/Ms, Bs/Ms;
    Ks/Mus, -Kus/Mus, Bs/Mus, -(Bs+Bus)/Mus];

% Control Input Matrix B (for Active Force u in Newtons)
B_rel = [0; 0; 1/Ms; -1/Mus];

% 3. LQR DESIGN (Based on Force)
Q = diag([100, 0, 800, 0]);
R = 0.05; % Penalty on Active Force

% Calculate LQR Gain Matrix K (Outputs Force)
[K_rel, ~, ~] = lqr(A_rel, B_rel, Q, R);

fprintf('LQR Gain Matrix K_rel:\n');
disp(K_rel);
